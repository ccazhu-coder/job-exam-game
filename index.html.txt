  <script>
    const API_URL = 'https://script.google.com/macros/s/AKfycbxPmwrWrAf7qkEHfxg4BBOtzJPkFBQFtd8ZB7lVSSWuECrduMPwZJ4RXCtzu-vTZilN9Q/exec';

    // --- ①【整段取代 QuizEngine】 ---
    const QuizEngine = {
      shuffle(a) {
        for (let i = a.length - 1; i > 0; i--) {
          const j = Math.floor(Math.random() * (i + 1));
          [a[i], a[j]] = [a[j], a[i]];
        }
        return a;
      },

      getRecentHistory() {
        return JSON.parse(localStorage.getItem("recent_q") || "[]");
      },

      saveHistory(ids) {
        let history = this.getRecentHistory();
        history = [...ids, ...history];
        history = [...new Set(history)].slice(0, 50);
        localStorage.setItem("recent_q", JSON.stringify(history));
      },

      generate(mode, db) {
        const recent = this.getRecentHistory();
        let pool = db.filter(q => !recent.includes(q.id));
        if (pool.length < 50) pool = db; // 題庫不足補題

        let selected = [];
        if (mode === "exam") {
          const singles = pool.filter(q => String(q.type).includes("單"));
          const multis = pool.filter(q => String(q.type).includes("複"));
          selected = [
            ...this.shuffle(singles).slice(0, 60),
            ...this.shuffle(multis).slice(0, 20)
          ];
        } else {
          selected = this.shuffle(pool).slice(0, 10);
        }

        this.saveHistory(selected.map(q => q.id));
        return this.shuffle(selected);
      }
    };

    const App = {
      state: { db: [], user: null, isVip: false, quiz: null, todayUsage: 0, limit: 30 },

      // --- ②【取代 App.init（加入防刷機制）】 ---
      async init() {
        const today = new Date().toDateString();
        const savedDate = localStorage.getItem('usage_date');

        if (savedDate !== today) {
          localStorage.setItem('usage_date', today);
          localStorage.setItem('usage_count', 0);
        }

        this.todayUsage = parseInt(localStorage.getItem('usage_count') || '0');
        this.updateLimitUI();

        try {
          const res = await fetch(API_URL);
          const result = await res.json();
          if (result.success) {
            this.state.db = result.data;
            UI.hideLoader();
          }
        } catch (err) { console.error("題庫同步失敗", err); }

        const saved = localStorage.getItem('user_p');
        if(saved) {
          this.state.user = JSON.parse(saved);
          this.state.isVip = this.state.user.isVip;
          this.syncUI();
        }
      },

      updateLimitUI() {
        const rem = Math.max(0, this.state.limit - this.todayUsage);
        document.getElementById('limitBadge').textContent = `今日剩餘: ${this.state.isVip ? '無限' : rem + ' 題'}`;
      },

      showView(id) {
        document.querySelectorAll('.view-section').forEach(v => v.classList.remove('active'));
        document.getElementById(id).classList.add('active');
        window.scrollTo(0,0);
      },

      async login() {
        const email = document.getElementById('authEmail').value;
        if(!email) return;
        try {
          const res = await fetch(API_URL, {
            method: 'POST',
            body: JSON.stringify({ action: 'login', email: email })
          });
          const r = await res.json();
          if(r.success) {
            this.state.user = r.user;
            this.state.isVip = r.user.isVip;
            localStorage.setItem('user_p', JSON.stringify(r.user));
            this.syncUI();
            UI.hideModal('authModal');
            this.showView('dashboardView');
          }
        } catch(e) { alert("登入失敗。"); }
      },

      syncUI() {
        if(this.state.user) {
          document.getElementById('btnAuth').style.display = 'none';
          document.getElementById('btnDash').style.display = 'inline-flex';
          document.getElementById('userName').textContent = this.state.user.name;
          document.getElementById('userRole').textContent = this.state.isVip ? "💎 VIP 尊榮學員" : "一般學員";
          document.getElementById('expBar').style.width = (this.state.user.exp % 100) + "%";
        }
        this.updateLimitUI();
      },

      // --- ③【取代 startMode（同步修正）】 ---
      async startMode(mode) {
        if (!this.state.isVip && this.todayUsage >= this.state.limit) {
          alert("今日免費額度已滿，請升級 VIP");
          this.showView('vipView');
          return;
        }

        if (mode === 'exam' && !this.state.isVip) {
          this.showView('vipView');
          return;
        }

        const qs = QuizEngine.generate(mode, this.state.db);

        this.state.quiz = {
          mode,
          qs,
          cur: 0,
          answers: {},
          results: {},
          startTime: Date.now()
        };

        this.showView('quizView');
        this.renderQ();
        this.startTimer();
      },

      renderQ() {
        const qz = this.state.quiz; const q = qz.qs[qz.cur];
        const isSub = !!qz.results[q.id];

        document.getElementById('quizCounter').textContent = `${qz.cur + 1} / ${qz.qs.length}`;
        document.getElementById('qTypeHint').textContent = String(q.type).includes("複") ? "💡【複選題】" : "💡【單選題】";
        document.getElementById('qText').textContent = q.question;

        const opts = ['A','B','C','D'].map(k => ({k, v: q[k] || q['option'+k]})).filter(x=>x.v);
        document.getElementById('optionsGrid').innerHTML = opts.map(o => {
          let cls = 'opt-card';
          const sel = (qz.answers[q.id] || []).includes(o.k);
          if (sel) cls += ' selected';
          if (isSub) {
            if (String(q.answer).includes(o.k)) cls += ' is-correct';
            else if (sel) cls += ' is-wrong';
          }
          return `<div class="${cls}" onclick="App.pick('${o.k}')"><b>${o.k}</b> ${o.v}</div>`;
        }).join('');

        document.getElementById('btnSubmit').style.display = isSub ? 'none' : 'inline-flex';
        document.getElementById('btnNext').style.display = (isSub && qz.cur < qz.qs.length - 1) ? 'inline-flex' : 'none';
        
        const panel = document.getElementById('analysisBox');
        if (isSub) {
          document.getElementById('analysisContent').innerHTML = `
            <p style="margin-bottom:1rem; color:#e2e8f0;">${q.ai_explanation || '解析同步中...'}</p>
            <p style="color:var(--warning); font-weight:700;">💡 記憶口訣：${q.memory || '多做幾次就會囉！'}</p>
          `;
          panel.style.display = 'block';
        } else { panel.style.display = 'none'; }

        document.getElementById('qNavGrid').innerHTML = qz.qs.map((_, i) => `
          <button class="qnav-btn ${i===qz.cur?'current':''} ${qz.results[qz.qs[i].id]?'done':''}" onclick="App.jump(${i})">${i+1}</button>
        `).join('');
      },

      pick(k) {
        const qz = this.state.quiz; const q = qz.qs[qz.cur];
        if (qz.results[q.id]) return;
        let cur = qz.answers[q.id] || [];
        if (String(q.type).includes("單")) cur = [k];
        else {
          if (cur.includes(k)) cur = cur.filter(x => x !== k);
          else cur.push(k);
        }
        qz.answers[q.id] = cur;
        this.renderQ();
      },

      // --- 🔥 ⑥【修正 submitAnswer（確保記錄）】 ---
      async submitAnswer() {
        const qz = this.state.quiz;
        const q = qz.qs[qz.cur];

        const ans = (qz.answers[q.id] || []).sort().join('');
        if (!ans) return;

        const isCorrect = ans === q.answer;
        qz.results[q.id] = { isCorrect };

        if (!this.state.isVip) {
          this.todayUsage++;
          localStorage.setItem('usage_count', this.todayUsage);
          this.updateLimitUI();
        }

        if(this.state.user) {
          fetch(API_URL, {
            method: 'POST',
            body: JSON.stringify({
              action: 'submitAnswer',
              uid: this.state.user.uid,
              qid: q.id,
              correct: isCorrect
            })
          });
        }

        this.renderQ();
        
        // 若是最後一題且已作答，自動顯示交卷邏輯
        if (qz.cur === qz.qs.length - 1) {
           setTimeout(() => this.finishQuiz(), 1500);
        }
      },

      nextQ() {
        if (this.state.quiz.cur < this.state.quiz.qs.length - 1) {
          this.state.quiz.cur++; this.renderQ();
        } else {
          this.finishQuiz();
        }
      },
      jump(i) { this.state.quiz.cur = i; this.renderQ(); },

      // --- ④【取代 finishQuiz（真實分數＋結果頁）】 ---
      finishQuiz() {
        const total = this.state.quiz.qs.length;
        const correct = Object.values(this.state.quiz.results)
          .filter(r => r.isCorrect).length;

        const score = Math.round((correct / total) * 100);

        document.getElementById('finalScore').textContent = score + " 分";
        document.getElementById('finalDetail').textContent = `答對 ${correct} / ${total}`;

        if(this.state.user) {
          fetch(API_URL, {
            method: 'POST',
            body: JSON.stringify({
              action: 'saveSession',
              uid: this.state.user.uid,
              mode: this.state.quiz.mode,
              score: score
            })
          });
        }

        this.showView('resultView');
      },

      startTimer() {
        if(this.timer) clearInterval(this.timer);
        this.timer = setInterval(() => {
          const sec = Math.floor((Date.now() - this.state.quiz.startTime)/1000);
          const m = String(Math.floor(sec/60)).padStart(2,'0');
          const s = String(sec%60).padStart(2,'0');
          document.getElementById('quizTimer').textContent = `${m}:${s}`;
        }, 1000);
      }
    };

    const UI = {
      showModal: (id) => document.getElementById(id).classList.add('active'),
      hideModal: (id) => document.getElementById(id).classList.remove('active'),
      hideLoader: () => document.getElementById('loader').style.display = 'none'
    };

    window.App = App;
    App.init();
  </script>
